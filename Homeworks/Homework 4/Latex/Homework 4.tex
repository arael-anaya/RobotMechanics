\section*{Problem 1}

\begin{statement}
Derive from the inverse twist problem equations (p.~34 of notes) the simplified forward and inverse
equations for the twist parameterization in the special case of parallel displacement $\dvec$ with the
rotation axis $\khat$.
\end{statement}

\begin{align}
\dvec &= \big((\bm{I}-\R)\skewx{\khat} + \theta\khat\khat^{\mathsf{T}}\big)\vvec, \label{eq:p1-forward}\\[4pt]
\vvec &= \Bigg(\frac{\sin\theta}{2(1-\cos\theta)}\bm{I} + \frac{2(1-\cos\theta)-\theta\sin\theta}{2\theta(1-\cos\theta)}\khat\khat^{\mathsf{T}} \notag\\
&\qquad{}- \tfrac{1}{2}\skewx{\khat}\Bigg)\dvec, \label{eq:p1-inverse}\\[4pt]
\R\khat&=\khat,\quad \R^{\mathsf T}\khat=\khat \ \Rightarrow\ \khat^{\mathsf T}(\bm I-\R)=\bm 0, \label{eq:p1-krot}\\[2pt]
\text{Since } \khat^{\mathsf T}(\bm I-\R)=\bm 0\ \text{by \eqref{eq:p1-krot}, } (\bm I-\R)\bm w&\perp\khat\ \ \forall\bm w \ \Rightarrow\ (\bm I-\R)\skewx{\khat}\vvec \perp \khat, \label{eq:p1-perp}\\[4pt]
\dvec\parallel\khat &\ \Rightarrow\ (\bm I-\R)\skewx{\khat}\vvec = \bm 0, \label{eq:p1-dpar}\\[2pt]
\text{With \eqref{eq:p1-dpar}, }\theta\neq0\ (\mathrm{mod}\ 2\pi)&:\ (\bm I-\R)\ \text{invertible on}\ \khat^{\perp},\ \text{and}\ \skewx{\khat}\vvec\in\khat^{\perp}\notag\\
&\ \Rightarrow\ \khat\times\vvec=\bm 0 \ \Rightarrow\ \vvec\parallel\khat, \label{eq:p1-vpar}\\[6pt]
\vvec&=v\khat,\quad \dvec=d\khat,\\[4pt]
d\khat &= \theta\khat\khat^{\mathsf T}(v\khat) = \theta v\khat,\\[2pt]
&\boxed{d=\theta v,} \label{eq:p1-fwd-final}\\[8pt]
v\khat &= \left[\frac{\sin\theta}{2(1-\cos\theta)}+\frac{2(1-\cos\theta)-\theta\sin\theta}{2\theta(1-\cos\theta)}\right]d\khat,\\[4pt]
\frac{\theta\sin\theta+2(1-\cos\theta)-\theta\sin\theta}{2\theta(1-\cos\theta)}&=\frac{2(1-\cos\theta)}{2\theta(1-\cos\theta)}=\frac{1}{\theta},\\[4pt]
&\boxed{v=\dfrac{d}{\theta}.} \label{eq:p1-inv-final}
\end{align}

\section*{Problem 6}

\begin{statement}
Derive the DH parameters for the RRPR manipulator shown in the figure below.
\begin{center}
    \includegraphics[width=.4\columnwidth]{Figures/SCARA_manipulator_siciliano_fig_2.36.png}
    \captionof{figure}{RRPR manipulator.}
\end{center}
\end{statement}

\textbf{Frame assignment:}
\begin{itemize}[leftmargin=1.6em, itemsep=1pt]
    \item Frame 0: $z_0$ up along the joint 1 axis; origin at arm height.
    \item Frame 1: $z_1$ up along the joint 2 axis (parallel to $z_0$); $x_1$ along link 1, from joint 1 to joint 2.
    \item Frame 2: $z_2$ down along the joint 3 (prismatic) axis; $x_2$ along link 2, from joint 2 to joint 3.
    \item Frame 3: $z_3$ down along the joint 4 axis (parallel to $z_2$); $x_3$ along $x_2$, since $a_3 = 0$.
    \item Frame 4: end-effector frame at the tip, $z_4$ down along $z_3$.
\end{itemize}

\textbf{Parameter definitions:}
\begin{itemize}[leftmargin=1.6em, itemsep=1pt]
    \item $a_1$: link 1 length, horizontal offset from the joint 1 axis to the joint 2 axis.
    \item $a_2$: link 2 length, horizontal offset from the joint 2 axis to the joint 3 axis.
    \item $d_1$: vertical offset from frame 0 to frame 1. We take frame 0 at arm height (not the base plate), so $d_1 = 0$, matching the standard textbook version of this manipulator.
    \item $d_4$: fixed vertical offset from frame 3 to the end-effector tip.
    \item $\alpha_2 = \pi$: $z_2$ (prismatic axis, pointing down) is antiparallel to $z_1$ (pointing up), a $\pi$ twist about $x_2$.
\end{itemize}

\begin{center}
\begin{tabular}{c | c | c | c | c}
\hline
 Link & $a$ & $d$ & $\alpha$ & $\theta$\\  \hline
 $1$ & $a_1$ & $0$  & 0 & $\star$ \\
 $2$ & $a_2$ & 0 & $\pi$ & $\star$ \\
 $3$ & 0 & $\star$ & 0 & 0 \\
 $4$ & 0 & $d_4$ & 0 & $\star$ \\ \hline
\end{tabular}
\end{center}
